import json
from pathlib import Path
from collections import defaultdict
from statistics import median


SOLD_PATH = Path(
    r"C:\Users\vpine\Desktop\RealEstate\gov_source_tests\wine_city_sold_5room_raw.json"
)

GIS_PATH = Path(
    r"C:\Users\vpine\Desktop\RealEstate\gabay_pricing_core\data\diagnostics\three_room_program_overlap_parcels.json"
)


def num(v):
    try:
        return float(v)
    except (TypeError, ValueError):
        return None


def integer(v):
    try:
        return int(float(v))
    except (TypeError, ValueError):
        return None


sold = json.loads(
    SOLD_PATH.read_text(encoding="utf-8")
)

gis = json.loads(
    GIS_PATH.read_text(encoding="utf-8")
)


# ============================================================
# Existing official program references
# ============================================================

official = {}

for p in gis.get("parcels", []):

    gush = integer(p.get("gush"))
    helka = integer(p.get("helka"))

    if gush is None or helka is None:
        continue

    prices = sorted({
        float(project["PriceForMeter"])
        for project in p.get("projects", [])
        if isinstance(
            project.get("PriceForMeter"),
            (int, float),
        )
        and project["PriceForMeter"] > 0
    })

    official[(gush, helka)] = {
        "prices": prices,
        "projects": p.get("projects", []),
    }


# ============================================================
# Exact 5-room local source, grouped by parcel
# ============================================================

by_parcel = defaultdict(list)

usable_rows = 0

for row in sold:

    if num(row.get("rooms")) != 5:
        continue

    gush = integer(row.get("gush"))
    helka = integer(row.get("helka"))

    area = num(row.get("area"))
    amount = num(row.get("dealAmount"))
    ppsm = num(row.get("pricePerSqm"))

    if (
        gush is None
        or helka is None
        or area is None
        or amount is None
    ):
        continue

    if ppsm is None and area > 0:
        ppsm = amount / area

    if ppsm is None:
        continue

    usable_rows += 1

    by_parcel[(gush, helka)].append({
        "date": row.get("dealDate"),
        "amount": amount,
        "area": area,
        "ppsm": ppsm,
        "address": row.get("address"),
        "floor": row.get("floor"),
        "assetId": row.get("assetId"),
        "lat": row.get("lat"),
        "lng": row.get("lng"),
    })

print("STANDARD 5-ROOM LOCAL SOLD EVIDENCE — PARCEL CHECK")
print("raw_rows:", len(sold))
print("usable_5room_rows:", usable_rows)
print("unique_parcels:", len(by_parcel))


covered = 0
not_covered = 0

for key in by_parcel:

    info = official.get(key)

    if info and info["projects"]:
        covered += 1
    else:
        not_covered += 1


print("parcels_with_existing_program_GIS_reference:", covered)
print("parcels_without_existing_program_GIS_reference:", not_covered)


# ============================================================
# Parcel-level structure
# ============================================================

    )

    areas = sorted(
        r["area"]
        for r in rows
    )

    info = official.get(

        info["prices"]

        else []
    )

    )

    print(

    print(
        "area_range:",

        "->",
        f"{max(areas):,.0f}",
        "m²"
    )

    print(
        "ppsm_range:",
        f"{min(prices):,.0f}",
        "->",
        f"{max(prices):,.0f}"
    )

    print(
        "ppsm_median:",

    )


    if official_prices:

        print(

                f"{x:,.0f}"

        )

                prices[i + 1],
            )

            )
        ]

        )
